/// <summary>

    /// Найти всех пациентов у которых есть приемы раньше 2017 года.

    /// Предложить, как минимум 3 решения.

    /// Измерить производительность.

    /// Выбрать лучшее, описать почему оно лучшее и в сравнении с чем.

    /// </summary>

    public class ReceptionsAndPatients

    {

        private const int YEAR = 2017;

  

        private IList<Reception> receptions;

        private IList<Patient> patients;

  

        public ReceptionsAndPatients()

        {

            var rnd = new Random();

  

            receptions = Enumerable.Range(1, 500_000)

                .SelectMany(pid => Enumerable.Range(1, rnd.Next(0, 100))

                .Select(rid => new Reception

                    {

                        PatientId = pid,

                        ReceptionStart = new DateTime(YEAR, 06, 03).AddDays(-rnd.Next(1, 500))

                    }))

                .ToList();

  

            patients = Enumerable.Range(1, 500_000)

                .Select(pid =>

                    new Patient

                    {

                        Id = pid,

                        Surname = $"Иванов{pid}"

                    })

                .ToList();

        }

  

        [Benchmark]

        public void Scenario_1()

        {

            patients.Where(p =>

                        receptions.Any(r => r.PatientId == p.Id

                                         && r.ReceptionStart.Year < YEAR));

        }

  

        [Benchmark]

        public void Scenario_2()

        {

            var filteredReceptions = receptions.Where(r => r.ReceptionStart.Year < YEAR);

            patients.Where(p => filteredReceptions.Any(fr => fr.PatientId == p.Id));

        }

  

        [Benchmark]

        public void Scenario_3()

        {

            var filteredReceptions =

                receptions.GroupBy(r => r.PatientId)

                          .ToDictionary(g => g.Key, g => g.ToList());

  

            patients.Where(p =>

                filteredReceptions.ContainsKey(p.Id)

             && filteredReceptions[p.Id].Any(r => r.ReceptionStart.Year < YEAR));

        }

    }

  

    public class Reception

    {

        public int PatientId { get; set; }

        public DateTime ReceptionStart { get; set; }

    }

  

    public class Patient

    {

        public int Id { get; set; }

        public string Surname { get; set; }

  

        public override string ToString()

        {

            return $"{Id} - {Surname}";

        }

    }
var rp = new ReceptionsAndPatients();

        Stopwatch sw = Stopwatch.StartNew();

  

        // Первый вариант

        sw.Restart();

        rp.Scenario_1();

        Console.WriteLine($"Решение 1 заняло: {sw.ElapsedMilliseconds} мс");

  

        // Второй вариант

        sw.Restart();

        rp.Scenario_2();

        Console.WriteLine($"Решение 2 заняло: {sw.ElapsedMilliseconds} мс");

  

        // Третий вариант

        sw.Restart();

        rp.Scenario_3();

        Console.WriteLine($"Решение 3 заняло: {sw.ElapsedMilliseconds} мс");
